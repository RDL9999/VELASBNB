// ⚙️ NÚCLEO DEL MODELO DIXON-COLES CON SEMILLA

let rng=()=>Math.random();

function setSeed(s){let a=s>>>0;rng=()=>(a|=0,a=a+0x6D2B79F5|0,((t=(t=Math.imul(a^a>>>15,1|a))+(Math.imul(t^t>>>7,61|t))^t,t^t>>>14)>>>0)/4294967296)}

function poisson(l){if(l<=0)return 0;let L=Math.exp(-l),k=0,p=1;do{k++;p*=rng()}while(p>L);return k-1}
function tau(x,y,l,m,r){if(x==0&&y==0)return 1-l*m*r;if(x==0&&y==1)return 1+l*r;if(x==1&&y==0)return 1+m*r;if(x==1&&y==1)return 1-r;return 1}
function sample(l,m,r){let gl,gv,t,g=0;do{gl=poisson(l);gv=poisson(m);t=tau(gl,gv,l,m,r);g++}while((t<=0||rng()>=Math.min(1,t))&&g<5000);return[gl,gv]}

function simular(){
// Leer inputs
let aL=+document.getElementById('aL').value,dL=+document.getElementById('dL').value
let aV=+document.getElementById('aV').value,dV=+document.getElementById('dV').value
let mu=+document.getElementById('mu').value,rho=+document.getElementById('rho').value
let g=+document.getElementById('gamma').value,n=+document.getElementById('N').value
let seed=+document.getElementById('seed').value
let nomL=document.getElementById('nomL').textContent,nomV=document.getElementById('nomV').textContent

// === SEMILLA FUNCIONAL ===
setSeed(seed);

// Calcular λ
let lL=aL*dV*g*mu,lV=aV*dL*mu
document.getElementById('preview').innerHTML=`<strong>λ ${nomL}:</strong> ${lL.toFixed(3)} · <strong>λ ${nomV}:</strong> ${lV.toFixed(3)}`

// Simular
let w1=0,dr=0,w2=0,btts=0,o15=0,o25=0,o35=0,tg=0,sumL=0,sumV=0,sc={}
for(let i=0;i<n;i++){
 let[gl,gv]=sample(lL,lV,rho)
 sumL+=gl;sumV+=gv;let t=gl+gv;tg+=t
 gl>gv?w1++:gl==gv?dr++:w2++
 if(gl>0&&gv>0)btts++
 if(t>1)o15++
 if(t>2)o25++
 if(t>3)o35++
 let k=gl+'-'+gv;sc[k]=(sc[k]||0)+1
}

// Formatear
let p=x=>((x/n*100).toFixed(1)+'%')
let ci=x=>`±${(1.96*Math.sqrt(x/n*(1-x/n)/n)*100).toFixed(1)}pp`

// Renderizar 1X2
document.getElementById('p1').innerHTML=`<div class="p-row"><span>Gana ${nomL}</span><span><span class="v">${p(w1)}</span><span class="ci">${ci(w1)}</span></span></div>`
document.getElementById('pX').innerHTML=`<div class="p-row"><span>Empate</span><span><span class="v">${p(dr)}</span><span class="ci">${ci(dr)}</span></span></div>`
document.getElementById('p2').innerHTML=`<div class="p-row"><span>Gana ${nomV}</span><span><span class="v">${p(w2)}</span><span class="ci">${ci(w2)}</span></span></div>`
document.getElementById('b1').style.width=w1/n*100+'%'
document.getElementById('bX').style.width=dr/n*100+'%'
document.getElementById('b2').style.width=w2/n*100+'%'

// Renderizar goles
document.getElementById('goles').innerHTML=
`<div class="p-row"><span>Ambos Marcan (Sí)</span><span><span class="v">${p(btts)}</span><span class="ci">${ci(btts)}</span></span></div>
<div class="p-row"><span>Más de 2.5</span><span><span class="v">${p(o25)}</span><span class="ci">${ci(o25)}</span></span></div>
<div class="p-row"><span>Menos de 2.5</span><span><span class="v">${p(n-o25)}</span><span class="ci">${ci(n-o25)}</span></span></div>
<div class="p-row"><span>Más de 3.5</span><span><span class="v">${p(o35)}</span><span class="ci">${ci(o35)}</span></span></div>`

// xG
document.getElementById('xg').textContent=`xG ${nomL}: ${(sumL/n).toFixed(2)} · xG ${nomV}: ${(sumV/n).toFixed(2)} · Promedio total: ${(tg/n).toFixed(2)}`

// Top 3
let top=Object.entries(sc).sort((a,b)=>b[1]-a[1]).slice(0,3)
let tb=document.getElementById('top3');tb.innerHTML=''
top.forEach((s,i)=>{tb.innerHTML+=`<tr class="${i==0?'rank1':''}"><td>${i+1}</td><td>${nomL.substring(0,4).toUpperCase()} ${s[0].replace('-',' - ')} ${nomV.substring(0,4).toUpperCase()}</td><td>${p(s[1])}</td></tr>`})

document.getElementById('res').style.display='block'
}
